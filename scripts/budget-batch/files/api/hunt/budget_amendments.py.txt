"""Operator-requested budget increases; never a new execution-authority path.

The Hunt row lock serializes extensions with admissions, settlement and cancellation.
Only ceilings change. Existing holds, actual usage, action snapshots, identities and
permissions are untouched. The small audit table is not another traffic ledger.
"""

from __future__ import annotations

from dataclasses import fields, replace
import hashlib
import json
from typing import Any, Literal, Mapping
import uuid

from fastapi import HTTPException
from pydantic import BaseModel, ConfigDict, Field, StrictInt

from .device_policy import DeviceHuntPolicyState
from .start_contract import HuntBudget, HuntStartPolicy, ZEROABLE_HUNT_BUDGET_DIMENSIONS


SCHEMA_VERSION = "hunt-budget-amendment/v1"
# JSON/JavaScript clients must represent a requested total exactly. This is a
# serialization boundary, not another preset that caps legitimate operator work.
MAX_BUDGET_VALUE = 2**53 - 1
BUDGET_FIELDS = frozenset(field.name for field in fields(HuntBudget))

BUDGET_AMENDMENT_SCHEMA_SQL = """
ALTER TABLE hunt_runs ADD COLUMN IF NOT EXISTS budget_revision INTEGER NOT NULL DEFAULT 0;
CREATE TABLE IF NOT EXISTS hunt_budget_amendments (
    id UUID PRIMARY KEY,
    hunt_run_id UUID NOT NULL REFERENCES hunt_runs(id) ON DELETE CASCADE,
    revision INTEGER NOT NULL CHECK (revision > 0),
    request_key_sha256 TEXT NOT NULL,
    request_digest TEXT NOT NULL,
    limits_before JSONB NOT NULL,
    limits_after JSONB NOT NULL,
    used_snapshot JSONB NOT NULL,
    reason TEXT NOT NULL,
    status_before TEXT NOT NULL,
    status_after TEXT NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE (hunt_run_id, revision),
    UNIQUE (hunt_run_id, request_key_sha256)
);
"""


class HuntBudgetAmendmentRequest(BaseModel):
    model_config = ConfigDict(extra="forbid")
    schema_version: Literal["hunt-budget-amendment/v1"] = SCHEMA_VERSION
    # Total limits, not increments: retrying a request must not double its effect.
    limits: dict[str, StrictInt] = Field(min_length=1, max_length=len(BUDGET_FIELDS))
    expected_revision: int = Field(ge=0, le=2**31 - 2, strict=True)
    idempotency_key: str = Field(min_length=1, max_length=128, pattern=r"^[A-Za-z0-9_.:-]+$")
    operator_confirmed: bool = Field(strict=True)
    resume: bool = Field(default=False, strict=True)
    reason: str = Field(default="Operator requested additional budget", max_length=500)


def _object(value: Any) -> dict[str, Any]:
    if isinstance(value, str):
        try:
            value = json.loads(value)
        except (TypeError, ValueError) as exc:
            raise HTTPException(409, "Saved Hunt budget context is unavailable") from exc
    if not isinstance(value, Mapping):
        raise HTTPException(409, "Saved Hunt budget context is unavailable")
    return dict(value)


def budget_limits_after(run: Mapping[str, Any], request: HuntBudgetAmendmentRequest) -> dict[str, int]:
    """Increase requested dimensions using the existing policy's zero-dimension rules."""
    if request.operator_confirmed is not True:
        raise HTTPException(422, "A budget increase must be explicitly requested by the operator")
    unknown = set(request.limits) - BUDGET_FIELDS
    if unknown:
        raise HTTPException(422, "Unknown Hunt budget dimensions: " + ", ".join(sorted(unknown)))
    before = _object(run["budget_json"])
    if set(before) != BUDGET_FIELDS or any(type(value) is not int or value < 0 for value in before.values()):
        raise HTTPException(409, "Saved Hunt budget is incomplete or invalid")
    permitted = frozenset(amendable_dimensions(run))
    after = dict(before)
    for name, value in request.limits.items():
        minimum = 0 if name in ZEROABLE_HUNT_BUDGET_DIMENSIONS else 1
        if not minimum <= value <= MAX_BUDGET_VALUE:
            raise HTTPException(422, f"{name} must be between {minimum} and {MAX_BUDGET_VALUE}")
        if value < before[name]:
            raise HTTPException(422, f"Budget amendments cannot decrease {name}")
        if name not in permitted and value > 0:
            raise HTTPException(403, f"{name} needs permission absent from this Hunt; a budget increase does not grant it")
        after[name] = value
    if after == before:
        raise HTTPException(422, "At least one total limit must increase")
    return after


def amendable_dimensions(run: Mapping[str, Any]) -> list[str]:
    """The UI and validator use the same permission-owned dimension rules."""
    policy = _object(run.get("policy_json") or {})
    saved_policy = HuntStartPolicy(**{
        field.name: policy[field.name] for field in fields(HuntStartPolicy) if field.name in policy
    })
    forbidden = saved_policy.forbidden_budget_dimensions(
        target_kind=str(run.get("target_kind") or "web"), credentials_requested=policy.get("credential_access") is True,
    )
    return sorted(BUDGET_FIELDS - forbidden)


def amendment_public(row: Mapping[str, Any]) -> dict[str, Any]:
    return {
        "amendment_id": str(row["id"]), "revision": row["revision"],
        "limits_before": _object(row["limits_before"]),
        "limits_after": _object(row["limits_after"]),
        "used_snapshot": _object(row["used_snapshot"]), "reason": row["reason"],
        "status_before": row["status_before"], "status_after": row["status_after"],
        "created_at": row["created_at"].isoformat(),
    }


async def read_amendments(conn: Any, hunt_id: Any, *, after_revision: int = 0, limit: int = 50) -> dict[str, Any]:
    rows = await conn.fetch(
        "SELECT * FROM hunt_budget_amendments WHERE hunt_run_id=$1 AND revision>$2 "
        "ORDER BY revision LIMIT $3", hunt_id, after_revision, limit + 1,
    )
    page = rows[:limit]
    return {
        "amendments": [amendment_public(row) for row in page],
        "has_more": len(rows) > limit,
        "next_revision": page[-1]["revision"] if len(rows) > limit else None,
    }


async def apply_budget_amendment(conn: Any, hunt_id: Any, request: HuntBudgetAmendmentRequest) -> dict[str, Any]:
    """Called inside a transaction. Never resets accounting or redispatches old work."""
    row = await conn.fetchrow("SELECT * FROM hunt_runs WHERE id=$1 FOR UPDATE", hunt_id)
    if row is None:
        raise HTTPException(404, "Hunt not found")
    run = dict(row)
    key = hashlib.sha256(request.idempotency_key.encode()).hexdigest()
    payload = request.model_dump(exclude={"idempotency_key"})
    digest = hashlib.sha256(json.dumps(payload, sort_keys=True, separators=(",", ":")).encode()).hexdigest()
    previous = await conn.fetchrow(
        "SELECT * FROM hunt_budget_amendments WHERE hunt_run_id=$1 AND request_key_sha256=$2", hunt_id, key,
    )
    if previous is not None:
        if previous["request_digest"] != digest:
            raise HTTPException(409, "This idempotency key belongs to a different budget amendment")
        # A retry after cancellation returns history, not permission to resume.
        return _response(run, previous, replayed=True)
    if run["status"] not in {"active", "awaiting_planner", "budget_exhausted"} or run.get("completed_at") is not None:
        raise HTTPException(409, "Only an unfinished Hunt can receive additional budget")
    revision = int(run.get("budget_revision") or 0)
    if revision != request.expected_revision:
        raise HTTPException(409, {"message": "Hunt budget changed; refresh before extending", "budget_revision": revision})
    after = budget_limits_after(run, request)
    used = _object(run["budget_used_json"])
    context = _object(run["context_pack"])
    status = run["status"]
    stop_reason = run.get("stop_reason")
    if request.resume and status == "budget_exhausted":
        dimension = str(stop_reason or "").removeprefix("budget_exhausted:")
        limits = {**HuntBudget(**after).ledger_limits(), "candidates": after["max_candidates"], "verifications": after["max_verifications"]}
        if dimension not in limits or int(used.get(dimension) or 0) >= limits[dimension]:
            raise HTTPException(409, "Increase the exhausted dimension before resuming this Hunt")
        status, stop_reason = "awaiting_planner", None
    if run["target_kind"] == "device":
        state = DeviceHuntPolicyState.from_mapping(context.get("device_policy_state") or {})
        updates = {}
        for budget_key, state_key in (("max_http_requests", "request_limit"), ("max_device_fragility_points", "fragility_limit")):
            if budget_key in request.limits:
                updates[state_key] = max(getattr(state, state_key), after[budget_key])
        # Preserve pacing, usage, operator pauses and health freezes, including while resuming.
        context["device_policy_state"] = replace(state, **updates).public_dict()
    try:
        from redaction import redact_sensitive
    except ModuleNotFoundError:
        from scanner.redaction import redact_sensitive
    reason = str(redact_sensitive(request.reason, redact_strings=True, scrub_text=True))
    # Keep the denormalized budget mirror coherent, without changing any permission.
    policy = _object(run["policy_json"])
    policy["budget"] = dict(after)
    event = await conn.fetchrow(
        "INSERT INTO hunt_budget_amendments (id,hunt_run_id,revision,request_key_sha256,request_digest,"
        "limits_before,limits_after,used_snapshot,reason,status_before,status_after) "
        "VALUES($1,$2,$3,$4,$5,$6::jsonb,$7::jsonb,$8::jsonb,$9,$10,$11) RETURNING *",
        uuid.uuid4(), hunt_id, revision + 1, key, digest, json.dumps(_object(run["budget_json"])),
        json.dumps(after), json.dumps(used), reason, run["status"], status,
    )
    updated = await conn.fetchrow(
        "UPDATE hunt_runs SET budget_json=$2::jsonb,budget_revision=$3,context_pack=$4::jsonb,"
        "status=$5,stop_reason=$6,policy_json=$7::jsonb,updated_at=NOW() WHERE id=$1 RETURNING *",
        hunt_id, json.dumps(after), revision + 1, json.dumps(context), status, stop_reason, json.dumps(policy),
    )
    return _response(dict(updated), event, replayed=False)


def _response(run: Mapping[str, Any], event: Mapping[str, Any], *, replayed: bool) -> dict[str, Any]:
    return {
        "schema_version": SCHEMA_VERSION, "hunt_id": str(run["id"]),
        "budget_revision": int(run.get("budget_revision") or 0),
        "budget": _object(run["budget_json"]), "budget_used": _object(run["budget_used_json"]),
        "status": run["status"], "stop_reason": run.get("stop_reason"),
        "amendment": amendment_public(event), "replayed": replayed,
        "execution_started": False, "authority_changed": False,
        "device_traffic_frozen": bool(_object(run["context_pack"]).get("device_policy_state", {}).get("traffic_frozen")),
    }
